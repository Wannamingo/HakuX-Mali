cmake_minimum_required(VERSION 3.20)
project(hakux_rop_pass_tests C)
set(CMAKE_C_STANDARD 11)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_executable(test_rop_pass test_rop_pass.c)
target_include_directories(test_rop_pass PRIVATE "${REPO}")
if(MSVC)
  target_compile_options(test_rop_pass PRIVATE /W4 /WX /UNDEBUG)
else()
  target_compile_options(test_rop_pass PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endif()
enable_testing()
add_test(NAME rop_pass_transitions COMMAND test_rop_pass)
