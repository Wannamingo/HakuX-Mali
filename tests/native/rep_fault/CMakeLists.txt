cmake_minimum_required(VERSION 3.20)
project(hakux_rep_fault_tests C)
set(CMAKE_C_STANDARD 11)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_executable(test_rep_fault test_rep_fault.c)
target_include_directories(test_rep_fault PRIVATE "${REPO}")
if(MSVC)
  target_compile_options(test_rep_fault PRIVATE /W4 /WX /UNDEBUG)
else()
  target_compile_options(test_rep_fault PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endif()
enable_testing()
add_test(NAME rep_fault_restart COMMAND test_rep_fault)
