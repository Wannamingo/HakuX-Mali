cmake_minimum_required(VERSION 3.20)
project(hakux_visibility_tests C)
set(CMAKE_C_STANDARD 11)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_executable(test_visibility test_visibility.c)
target_include_directories(test_visibility PRIVATE "${REPO}")
if(MSVC)
  target_compile_options(test_visibility PRIVATE /W4 /WX /UNDEBUG)
else()
  target_compile_options(test_visibility PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endif()
enable_testing()
add_test(NAME unavailable_visibility COMMAND test_visibility)
