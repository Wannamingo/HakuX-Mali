cmake_minimum_required(VERSION 3.20)
project(hakux_vertex_repack_tests C)
set(CMAKE_C_STANDARD 11)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_executable(test_vertex_repack test_vertex_repack.c)
target_include_directories(test_vertex_repack PRIVATE "${REPO}")
if(MSVC)
  target_compile_options(test_vertex_repack PRIVATE /W4 /WX /UNDEBUG)
else()
  target_compile_options(test_vertex_repack PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endif()
enable_testing()
add_test(NAME vertex_repack COMMAND test_vertex_repack)
