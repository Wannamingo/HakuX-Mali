cmake_minimum_required(VERSION 3.20)
project(hakux_seh_tests C)
set(CMAKE_C_STANDARD 11)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_executable(test_seh test_seh.c)
target_include_directories(test_seh PRIVATE "${REPO}")
if(MSVC)
  target_compile_options(test_seh PRIVATE /W4 /WX /wd4100 /wd4204 /UNDEBUG)
else()
  target_compile_options(test_seh PRIVATE -Wall -Wextra -Werror -Wno-unused-parameter -UNDEBUG)
endif()
enable_testing()
add_test(NAME seh_access_violation COMMAND test_seh)
