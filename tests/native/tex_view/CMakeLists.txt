cmake_minimum_required(VERSION 3.20)
project(hakux_tex_view_tests C)
set(CMAKE_C_STANDARD 11)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_executable(test_tex_view test_tex_view.c)
target_include_directories(test_tex_view PRIVATE "${REPO}")
if(MSVC)
  target_compile_options(test_tex_view PRIVATE /W4 /WX /UNDEBUG)
else()
  target_compile_options(test_tex_view PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endif()
enable_testing()
add_test(NAME texture_view_aliases COMMAND test_tex_view)
