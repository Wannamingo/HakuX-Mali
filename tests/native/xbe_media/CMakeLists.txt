cmake_minimum_required(VERSION 3.20)
project(hakux_xbe_media_tests C)
set(CMAKE_C_STANDARD 11)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_executable(test_xbe_media test_xbe_media.c "${REPO}/native/core/xbe_media.c")
target_include_directories(test_xbe_media PRIVATE "${REPO}")
if(MSVC)
  target_compile_options(test_xbe_media PRIVATE /W4 /WX /UNDEBUG)
else()
  target_compile_options(test_xbe_media PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endif()
enable_testing()
add_test(NAME xbe_file_media COMMAND test_xbe_media)
